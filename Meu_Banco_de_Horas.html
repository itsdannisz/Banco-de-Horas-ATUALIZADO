Configurar a nuvem com Apps Script
  1. No Drive, crie um arquivo JSON com o conteúdo {"e":[],"rs":{}} e copie o ID da URL.
  2. Crie um projeto em script.google.com e cole o código abaixo.
  3. Nas propriedades do script, configure BH_FILE_ID com o ID do arquivo e BH_PASSWORD com uma senha forte.
  4. Implante como aplicativo da Web, executando como você e com acesso “Qualquer pessoa”. Autorize o acesso ao seu arquivo.
  5. Cole a URL terminada em /exec acima. Para atualizar uma implantação existente, publique uma nova versão.
O script verifica a senha e combina alterações com uma trava de gravação.
// Banco de horas — protocolo 3. Configure BH_FILE_ID e BH_PASSWORD nas propriedades do script.
function makeCore() {
  const fields = ['c1','c2','c3','c4','s1','s2','s3','x1','a1','a2','i1','i2','p1'];
  const config = ['saida','entrada','intervalo'];
  const clone = x => JSON.parse(JSON.stringify(x));
  const plain = x => x !== null && typeof x === 'object' && !Array.isArray(x);
  function date(s) {
    if (typeof s !== 'string' || !/^\d{4}-\d{2}-\d{2}$/.test(s)) return false;
    const [y,m,d] = s.split('-').map(Number), x = new Date(Date.UTC(y,m-1,d));
    return y >= 1900 && y <= 2200 && x.getUTCFullYear() === y && x.getUTCMonth() === m-1 && x.getUTCDate() === d;
  }
  const time = s => typeof s === 'string' && /^([01]\d|2[0-3]):[0-5]\d$/.test(s);
  const minutes = s => { if (!time(s)) throw Error('Horário inválido'); const [h,m]=s.split(':').map(Number); return h*60+m; };
  const fmt = (m,sign=false) => { if (!Number.isInteger(m)) return '—'; return (m<0?'-':sign&&m>0?'+':'')+Math.floor(Math.abs(m)/60)+'h'+String(Math.abs(m)%60).padStart(2,'0'); };
  const norm = s => String(s).normalize('NFD').replace(/[\u0300-\u036f]/g,'').toLowerCase();
  const stamp = v => Array.isArray(v) && v.length===2 && Number.isSafeInteger(v[0]) && v[0]>=0 && typeof v[1]==='string' && v[1].length<=100 ? [v[0],v[1]] : [0,''];
  const cmp = (a,b) => { a=stamp(a);b=stamp(b);return a[0]-b[0] || a[1].localeCompare(b[1]); };
  const eq = (a,b) => cmp(a,b)===0;
  const hash = s => {let h=2166136261;for(let i=0;i<s.length;i++)h=Math.imul(h^s.charCodeAt(i),16777619);return (h>>>0).toString(36);};
  const text = (s,max=20000) => {if(typeof s!=='string'||s.length>max)throw Error('Texto inválido ou muito longo');return s;};
  const idOK = s => typeof s==='string' && /^[\w:|.-]{1,240}$/.test(s) && !['__proto__','constructor','prototype'].includes(s);
  function empty() { return {version:3,saida:'18:00',entrada:'08:00',intervalo:60,e:[],rs:{},deleted:{entries:{},reports:{},conflicts:{}},cv:{},conflicts:{},t:0}; }
  function entry(x,i,legacy) {
    if(!plain(x)||!date(x.d)||!['extra','uso'].includes(x.t)||!Number.isInteger(x.m)||Math.abs(x.m)>10080) throw Error('Registro de horas inválido');
    if(x.t==='uso'&&x.m>=0)throw Error('Desconto inválido');
    const out={id:x.id|| (x.t==='extra'?'day:'+x.d:'old:'+hash(JSON.stringify([x.d,x.t,x.m,x.n||'',i]))),d:x.d,t:x.t,s:x.s||'',m:x.m,n:text(x.n||'',500),_v:stamp(x._v||legacy),_base:stamp(x._base)};
    if(!idOK(out.id)||typeof out.s!=='string'||(out.s!==''&&!time(out.s)))throw Error('Identificador ou saída inválidos');
    if(x.normal!==undefined){if(!time(x.normal))throw Error('Horário normal inválido');out.normal=x.normal;}
    if(x.next!==undefined){if(typeof x.next!=='boolean')throw Error('Virada de dia inválida');out.next=x.next;}
    if(x.start){if(!time(x.start))throw Error('Entrada inválida');out.start=x.start;out.break=x.break;out.expected=x.expected;
      if(!Number.isInteger(out.break)||out.break<0||out.break>720||!Number.isInteger(out.expected)||out.expected<0||out.expected>1440)throw Error('Jornada inválida');}
    if(out.t==='extra'&&out.normal&&out.s){const calculated=variation({exit:out.s,normal:out.normal,next:out.next||false,start:out.start||'',interval:out.break||0,expected:out.expected||0});if(calculated!==out.m)throw Error('Saldo inconsistente com os horários do registro');}
    return out;
  }
  function validate(raw) {
    if(!plain(raw)||!Array.isArray(raw.e)||raw.e.length>50000||!plain(raw.rs||{}))throw Error('Estrutura de backup inválida');
    if(raw.version!==undefined&&raw.version!==3)throw Error('Versão de backup não suportada');
    if(Object.keys(raw.rs||{}).length>50000)throw Error('Relatórios em excesso');
    const out=empty(), legacy=[Number.isSafeInteger(raw.t)&&raw.t>=0?raw.t:0,'legacy'];
    if(raw.saida!==undefined){if(!time(raw.saida))throw Error('Saída normal inválida');out.saida=raw.saida;}
    if(raw.entrada!==undefined){if(!time(raw.entrada))throw Error('Entrada normal inválida');out.entrada=raw.entrada;}
    if(raw.intervalo!==undefined){if(!Number.isInteger(raw.intervalo)||raw.intervalo<0||raw.intervalo>720)throw Error('Intervalo inválido');out.intervalo=raw.intervalo;}
    out.t=legacy[0];out.e=raw.e.map((x,i)=>entry(x,i,legacy));
    const ids=new Set();for(const e of out.e){if(ids.has(e.id))throw Error('Registro duplicado no backup');ids.add(e.id);out.t=Math.max(out.t,e._v[0]);}
    for(const [d,r] of Object.entries(raw.rs||{})){
      if(!date(d)||!plain(r)||!plain(r.f))throw Error('Relatório inválido');
      const rr={f:{},v:{},base:{}};for(const k of fields)if(r.f[k]!==undefined){rr.f[k]=text(r.f[k]);if(k==='c3'&&rr.f[k]!==''&&(!/^\d+$/.test(rr.f[k])||!Number.isSafeInteger(Number(rr.f[k]))))throw Error('Quantidade de leads inválida');rr.v[k]=stamp(r.v&&r.v[k]||legacy);rr.base[k]=stamp(r.base&&r.base[k]);out.t=Math.max(out.t,rr.v[k][0]);}out.rs[d]=rr;
    }
    for(const k of config){out.cv[k]=stamp(raw.cv&&raw.cv[k]||legacy);out.t=Math.max(out.t,out.cv[k][0]);}
    for(const group of ['entries','reports','conflicts']){
      const map=raw.deleted&&raw.deleted[group]||{};if(!plain(map)||Object.keys(map).length>100000)throw Error('Histórico de exclusões inválido');
      for(const [id,v] of Object.entries(map)){if(!(group==='reports'?date(id):idOK(id)))throw Error('Exclusão inválida');out.deleted[group][id]=stamp(v);out.t=Math.max(out.t,stamp(v)[0]);}
    }
    if(raw.conflicts!==undefined&&(!plain(raw.conflicts)||Object.keys(raw.conflicts).length>10000))throw Error('Conflitos inválidos');
    for(const [id,c] of Object.entries(raw.conflicts||{})){
      if(!idOK(id)||!plain(c)||!date(c.d)||!['entry','field'].includes(c.kind))throw Error('Conflito inválido');
      if(c.kind==='entry')out.conflicts[id]={id,kind:c.kind,d:c.d,alternative:entry(c.alternative,0,legacy),_v:stamp(c._v)};
      else {if(!fields.includes(c.key))throw Error('Campo inválido');out.conflicts[id]={id,kind:c.kind,d:c.d,key:c.key,alternative:text(c.alternative),_v:stamp(c._v)};}
      out.t=Math.max(out.t,stamp(c._v)[0]);
    }
    return prune(out);
  }
  function prune(s) {
    s.e=s.e.filter(e=>!s.deleted.entries[e.id]||cmp(e._v,s.deleted.entries[e.id])>0).sort((a,b)=>a.d.localeCompare(b.d)||a.id.localeCompare(b.id));
    for(const [d,r] of Object.entries(s.rs)){
      for(const k of Object.keys(r.f))if(s.deleted.reports[d]&&cmp(r.v[k],s.deleted.reports[d])<=0){delete r.f[k];delete r.v[k];delete r.base[k];}
      if(!Object.keys(r.f).length)delete s.rs[d];
    }
    for(const [id,c]of Object.entries(s.conflicts))if(s.deleted.conflicts[id]&&cmp(c._v,s.deleted.conflicts[id])<=0)delete s.conflicts[id];
    return s;
  }
  function merge(a,b) {
    a=validate(a);b=validate(b);const out=clone(a);out.t=Math.max(a.t,b.t);
    for(const k of config)if(cmp(b.cv[k],a.cv[k])>0){out[k]=b[k];out.cv[k]=b.cv[k];}
    for(const group of ['entries','reports','conflicts'])for(const [id,v]of Object.entries(b.deleted[group]))if(!out.deleted[group][id]||cmp(v,out.deleted[group][id])>0)out.deleted[group][id]=v;
    out.conflicts={...a.conflicts,...b.conflicts};
    const records=new Map(a.e.map(e=>[e.id,e]));
    for(const incoming of b.e){const old=records.get(incoming.id);if(!old){records.set(incoming.id,incoming);continue;}
      const winner=cmp(old._v,incoming._v)>=0?old:incoming, loser=winner===old?incoming:old;
      if(!eq(old._v,incoming._v)&&old._v[1]!==incoming._v[1]&&eq(old._base,incoming._base)&&JSON.stringify({...old,_v:0})!==JSON.stringify({...incoming,_v:0})){
        const versions=[old._v,incoming._v].map(x=>x.join('.')).sort().join('|'),id='ce:'+hash(incoming.id+versions);
        out.conflicts[id]={id,kind:'entry',d:incoming.d,alternative:loser,_v:winner._v};
      }records.set(incoming.id,winner);
    }out.e=[...records.values()];
    for(const [d,r]of Object.entries(b.rs)){
      const dest=out.rs[d]||(out.rs[d]={f:{},v:{},base:{}});
      for(const k of Object.keys(r.f)){
        if(dest.f[k]!==undefined&&!eq(dest.v[k],r.v[k])&&dest.v[k][1]!==r.v[k][1]&&eq(dest.base[k],r.base[k])&&dest.f[k]!==r.f[k]){
          const win=cmp(dest.v[k],r.v[k])>=0,versions=[dest.v[k],r.v[k]].map(x=>x.join('.')).sort().join('|'),id='cf:'+hash(d+k+versions);
          out.conflicts[id]={id,kind:'field',d,key:k,alternative:win?r.f[k]:dest.f[k],_v:win?dest.v[k]:r.v[k]};
        }
        if(dest.f[k]===undefined||cmp(r.v[k],dest.v[k])>0){dest.f[k]=r.f[k];dest.v[k]=r.v[k];dest.base[k]=r.base[k];}
      }
    }
    return prune(out);
  }
  function variation({exit,normal,next=false,start='',interval=0,expected=0}) {
    const end=minutes(exit)+(next?1440:0);
    if(start){const span=end-minutes(start);if(span<0||span>1440||interval>span)throw Error('Confira entrada, saída, intervalo e dia seguinte');return span-interval-expected;}
    const m=end-minutes(normal);if(m>1440)throw Error('Confira a saída no dia seguinte');return m;
  }
  function parseCSV(t) {
    t=String(t).replace(/^\uFEFF/,'');if(!t.trim())throw Error('O CSV está vazio');
    const first=t.split(/\r?\n/)[0];let quotes=false,commas=0,semis=0;for(let i=0;i<first.length;i++){if(first[i]==='"')quotes=!quotes;else if(!quotes){if(first[i]===',')commas++;if(first[i]===';')semis++;}}
    const delim=semis>commas?';':',',rows=[];let row=[],value='',quoted=false;
    for(let i=0;i<t.length;i++){const x=t[i];if(quoted){if(x==='"'){if(t[i+1]==='"'){value+='"';i++;}else quoted=false;}else value+=x;}
      else if(x==='"'){if(value!=='')throw Error('Aspas inválidas no CSV');quoted=true;}
      else if(x===delim){row.push(value);value='';}
      else if(x==='\n'||x==='\r'){if(x==='\r'&&t[i+1]==='\n')i++;row.push(value);rows.push(row);row=[];value='';}else value+=x;}
    if(quoted)throw Error('Aspas não fechadas no CSV');if(value||row.length){row.push(value);rows.push(row);}
    return rows.filter(r=>r.some(x=>x.trim()));
  }
  function numeric(v,locale='pt') {
    let s=String(v||'').trim().replace(/[^\d.,-]/g,'');if(!s)return null;
    if(locale==='pt'){if(s.includes(','))s=s.replace(/\./g,'').replace(',','.');else if(/^\d{1,3}(\.\d{3})+$/.test(s))s=s.replace(/\./g,'');}
    else s=s.replace(/,/g,'');const n=Number(s);return Number.isFinite(n)&&n>=0?n:null;
  }
  function metaCSV(t,reportDate) {
    const rows=parseCSV(t),h=rows[0].map(norm),ix=re=>h.findIndex(x=>re.test(x));
    const i={name:ix(/nome da campanha|campaign name/),result:ix(/^resultados$|^results$/),type:ix(/tipo de resultado|indicador de resultado|result type|result indicator/),reach:ix(/alcance|reach/),spent:ix(/valor usado|amount spent|gasto/),cost:ix(/custo por resultado|cost per result/),from:ix(/inicio dos relatorios|inicio do relatorio|reporting starts/),to:ix(/termino dos relatorios|fim dos relatorios|reporting ends/)};
    if(i.name<0)throw Error('Não achei a coluna Nome da campanha');
    const locale=h.some(x=>/campaign name/.test(x))?'en':'pt',data=rows.slice(1).filter(r=>(r[i.name]||'').trim());if(!data.length)throw Error('O CSV não contém campanhas');
    const campaigns=new Set(),stats=[],warnings=new Set();let leads=0,validLeadRows=0;
    for(const r of data){if(r.length!==h.length)throw Error('Há linhas com quantidade incorreta de colunas');campaigns.add(r[i.name].trim());const bits=[];
      if(i.reach>=0&&r[i.reach])bits.push('alcance '+r[i.reach]);if(i.result>=0&&r[i.result])bits.push(r[i.result]+' resultados'+(i.type>=0&&r[i.type]?' ('+r[i.type]+')':''));
      if(i.spent>=0&&r[i.spent])bits.push('investido '+(locale==='pt'?'R$ ':'')+r[i.spent]);if(i.cost>=0&&r[i.cost])bits.push('custo/resultado '+r[i.cost]);stats.push(r[i.name].trim()+': '+(bits.join(', ')||'sem métricas'));
      const type=i.type>=0?norm(r[i.type]):'';if(i.result>=0&&/lead/.test(type)){const n=numeric(r[i.result],locale);if(n!==null&&Number.isSafeInteger(n)&&Number.isSafeInteger(leads+n)){leads+=n;validLeadRows++;}else warnings.add('Há uma quantidade de leads inválida; confira o CSV');}
      if(i.from>=0&&r[i.from]!==reportDate||i.to>=0&&r[i.to]!==reportDate)warnings.add('O período exportado difere da data do relatório');
    }
    if(!validLeadRows)warnings.add('Leads não identificados: confira o tipo de resultado e informe a quantidade manualmente');
    if(i.from<0||i.to<0)warnings.add('O arquivo não informa um período completo; confira a data da exportação');
    return {fields:{c1:[...campaigns].join(', '),c2:stats.join(' / '),c3:validLeadRows?String(leads):''},warnings:[...warnings],count:data.length};
  }
  return {fields,config,clone,empty,validate,merge,prune,stamp,cmp,date,time,minutes,fmt,norm,variation,parseCSV,numeric,metaCSV};
}

function doPost(e) {
  var lock=LockService.getScriptLock();
  try {
    var props=PropertiesService.getScriptProperties(), password=props.getProperty('BH_PASSWORD'), fileID=props.getProperty('BH_FILE_ID');
    if(!password||!fileID)return jsonResponse({protocol:3,erro:'Configure BH_FILE_ID e BH_PASSWORD'});
    if(!e||!e.postData||e.postData.contents.length>20000000)return jsonResponse({protocol:3,erro:'Pedido inválido'});
    var request=JSON.parse(e.postData.contents);
    if(request.k!==password)return jsonResponse({protocol:3,erro:'Senha incorreta'});
    if(request.op!=='sync')return jsonResponse({protocol:3,erro:'Operação inválida'});
    var core=makeCore(),incoming=core.validate(request.d);

    var stored=raw.trim()?core.validate(JSON.parse(raw)):core.empty();
    var combined=core.merge(stored,incoming);
    file.setContent(JSON.stringify(combined));
    return jsonResponse({protocol:3,ok:true,d:combined});
  } catch(error) { return jsonResponse({protocol:3,erro:'Falha ao validar ou gravar os dados. Confira o JSON e a configuração.'}); }
  finally { if(lock.hasLock())lock.releaseLock(); }
}
